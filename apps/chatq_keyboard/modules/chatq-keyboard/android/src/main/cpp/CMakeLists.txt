cmake_minimum_required(VERSION 3.22.1)
project(chatqpinyin LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(PINYIN ${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/pinyin)
file(GLOB CORE CONFIGURE_DEPENDS ${PINYIN}/jni/share/*.cpp)
add_library(chatqpinyin SHARED ${CORE} ${PINYIN}/chatq_decoder.cpp pinyin_jni.cpp)
target_include_directories(chatqpinyin PRIVATE ${PINYIN})
target_link_libraries(chatqpinyin log)
target_link_options(chatqpinyin PRIVATE "-Wl,-z,max-page-size=16384")
